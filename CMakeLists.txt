cmake_minimum_required(VERSION 3.21)
option(XASTIR_BUILD_QT "Build the Qt frontend" ON)
project(XastirQt VERSION 0.1.0 LANGUAGES C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)

if(XASTIR_BUILD_QT)
  enable_language(CXX)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network PrintSupport SerialPort)
  set(XASTIR_SQLITE_AMALGAMATION "" CACHE FILEPATH "Optional sqlite3.c source for static packaging")
  if(XASTIR_SQLITE_AMALGAMATION)
    get_filename_component(sqlite_include "${XASTIR_SQLITE_AMALGAMATION}" DIRECTORY)
    add_library(xastir_sqlite STATIC "${XASTIR_SQLITE_AMALGAMATION}")
    set_target_properties(xastir_sqlite PROPERTIES AUTOMOC OFF)
    target_include_directories(xastir_sqlite PUBLIC "${sqlite_include}")
    target_compile_definitions(xastir_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION)
    find_package(Threads REQUIRED)
    target_link_libraries(xastir_sqlite PUBLIC Threads::Threads)
    add_library(SQLite::SQLite3 ALIAS xastir_sqlite)
  else()
    find_package(SQLite3 3.24 REQUIRED)
  endif()
endif()
include(CTest)
add_library(xastir_core STATIC src/ax25_core.c src/kiss_core.c src/object_core.c src/aprs_position.c src/aprs_mice.c src/station_db.c src/message_store.c src/message_core.c src/station_view.c src/vector_map.c)
set_target_properties(xastir_core PROPERTIES AUTOMOC OFF C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(xastir_core PUBLIC ${PROJECT_SOURCE_DIR}/src)
if(NOT MSVC)
  target_compile_options(xastir_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(xastir_core PUBLIC m)
endif()
option(XASTIR_WITH_SHAPELIB "Enable basic shapefile vector maps when Shapelib is available" ON)
if(XASTIR_WITH_SHAPELIB)
  find_path(SHAPELIB_INCLUDE_DIR shapefil.h)
  find_library(SHAPELIB_LIBRARY NAMES shp shapelib)
  if(SHAPELIB_INCLUDE_DIR AND SHAPELIB_LIBRARY)
    target_include_directories(xastir_core PRIVATE ${SHAPELIB_INCLUDE_DIR})
    target_link_libraries(xastir_core PUBLIC ${SHAPELIB_LIBRARY})
    target_compile_definitions(xastir_core PRIVATE XASTIR_HAVE_SHAPELIB)
  else()
    message(STATUS "Shapelib unavailable: bundled worldhi.map remains supported")
  endif()
endif()
if(XASTIR_BUILD_QT)
  add_subdirectory(qt)
endif()

if(BUILD_TESTING)
  add_executable(test_vector_map tests/core/test_vector_map.c)
  set_target_properties(test_vector_map PROPERTIES AUTOMOC OFF)
  target_link_libraries(test_vector_map PRIVATE xastir_core)
  target_compile_definitions(test_vector_map PRIVATE VECTOR_DATA_DIR="${PROJECT_SOURCE_DIR}")
  add_test(NAME vector_map_core COMMAND test_vector_map)
  add_executable(test_aprs_position tests/core/test_aprs_position.c)
  set_target_properties(test_aprs_position PROPERTIES AUTOMOC OFF)
  target_link_libraries(test_aprs_position PRIVATE xastir_core)
  add_test(NAME aprs_position_core COMMAND test_aprs_position)
  foreach(component IN ITEMS ax25_core kiss_core object_core aprs_mice station_db message_core station_view)
    add_executable(test_${component} tests/core/test_${component}.c)
    set_target_properties(test_${component} PROPERTIES AUTOMOC OFF)
    target_link_libraries(test_${component} PRIVATE xastir_core)
    add_test(NAME ${component}_core COMMAND test_${component})
  endforeach()
endif()
